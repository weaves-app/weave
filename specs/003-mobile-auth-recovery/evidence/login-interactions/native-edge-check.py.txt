from pathlib import Path
import subprocess,xml.etree.ElementTree as ET,re,json
adb='/Users/pravinraj/Library/Android/sdk/platform-tools/adb'
def run(*args):return subprocess.check_output([adb,'-s','emulator-5554',*args])
def snapshot(name):
 run('shell','uiautomator','dump','/data/local/tmp/weave-method.xml')
 p=Path('/private/tmp/weave-method-'+name+'.xml');p.write_bytes(run('exec-out','cat','/data/local/tmp/weave-method.xml'))
 return ET.parse(p)
def node(tree,label):
 nodes=list(tree.iter('node'))
 match=next((n for n in nodes if n.get('content-desc')==label),None)
 return match if match is not None else next(n for n in nodes if n.get('text')==label)
def tap(tree,label):
 x1,y1,x2,y2=map(int,re.findall(r'\d+',node(tree,label).get('bounds')))
 run('shell','input','tap',str((x1+x2)//2),str((y1+y2)//2))
